"""
PROFITABILITY-FOCUSED PORTFOLIO STRATEGY
=========================================

Uses the existing out-of-sample predictions produced by the
classical ML pipeline.

Strategy:
    1. Ensemble the regression predictions
    2. Ensemble directional predictions
    3. Apply confidence filtering
    4. Apply technical-style confirmation using model agreement
    5. Rank stocks by expected return
    6. Select only the strongest opportunities
    7. Volatility-aware position sizing
    8. Limit portfolio concentration
    9. Apply stop-loss / take-profit logic
    10. Calculate portfolio performance and risk metrics

IMPORTANT:
This does NOT alter the trained ML models.
It operates on the existing test predictions.
"""

from __future__ import annotations

import warnings
from pathlib import Path

import numpy as np
import pandas as pd

warnings.filterwarnings("ignore")


# ============================================================
# PATHS
# ============================================================

PROJECT_ROOT = Path(__file__).resolve().parents[1]

REPORT_DIR = PROJECT_ROOT / "reports"

PREDICTIONS_FILE = REPORT_DIR / "predictions.parquet"

OUTPUT_STOCK = REPORT_DIR / "profit_strategy_stock_results.csv"
OUTPUT_DAILY = REPORT_DIR / "profit_strategy_daily_results.csv"
OUTPUT_METRICS = REPORT_DIR / "profit_strategy_metrics.csv"
OUTPUT_SIGNALS = REPORT_DIR / "profit_strategy_signals.csv"


# ============================================================
# STRATEGY CONFIGURATION
# ============================================================

INITIAL_CAPITAL = 100_000.0

# Transaction cost per complete trade.
TRANSACTION_COST = 0.0005

# Number of stocks held simultaneously.
MAX_POSITIONS = 8

# Minimum number of models predicting UP.
MIN_MODEL_AGREEMENT = 4

# Minimum ensemble directional confidence.
MIN_CONFIDENCE = 0.60

# Minimum predicted next-day return.
MIN_EXPECTED_RETURN = 0.003

# Maximum weight of one stock.
MAX_POSITION_WEIGHT = 0.20

# Minimum position weight.
MIN_POSITION_WEIGHT = 0.05

# Risk scaling.
VOLATILITY_LOOKBACK = 20

# Stop loss and take profit.
STOP_LOSS = 0.03
TAKE_PROFIT = 0.06


# ============================================================
# MODEL COLUMNS
# ============================================================

REGRESSION_MODELS = [
    "random_forest",
    "extra_trees",
    "hist_gradient_boosting",
    "xgboost",
    "lightgbm",
]

DIRECTION_MODELS = [
    "random_forest",
    "extra_trees",
    "hist_gradient_boosting",
    "xgboost",
    "lightgbm",
]


# ============================================================
# LOAD DATA
# ============================================================

def load_predictions() -> pd.DataFrame:

    print("=" * 70)
    print("LOADING EXISTING MODEL PREDICTIONS")
    print("=" * 70)

    if not PREDICTIONS_FILE.exists():
        raise FileNotFoundError(
            f"Predictions file not found:\n{PREDICTIONS_FILE}"
        )

    df = pd.read_parquet(PREDICTIONS_FILE)

    print(f"Rows loaded : {len(df):,}")
    print(f"Columns     : {len(df.columns)}")

    required = [
        "ticker",
        "date",
        "close",
        "target_return",
        "target_direction",
    ]

    for model in REGRESSION_MODELS:
        required.append(f"pred_return_{model}")

    for model in DIRECTION_MODELS:
        required.append(f"pred_direction_{model}")

    missing = [
        col for col in required
        if col not in df.columns
    ]

    if missing:
        raise ValueError(
            "Missing required columns:\n"
            + "\n".join(missing)
        )

    df["date"] = pd.to_datetime(df["date"])

    df = df.sort_values(
        ["date", "ticker"]
    ).reset_index(drop=True)

    return df


# ============================================================
# ENSEMBLE SIGNAL
# ============================================================

def create_ensemble_signal(
    df: pd.DataFrame,
) -> pd.DataFrame:

    df = df.copy()

    # --------------------------------------------------------
    # Regression ensemble
    # --------------------------------------------------------

    regression_columns = [
        f"pred_return_{model}"
        for model in REGRESSION_MODELS
    ]

    df["ensemble_return"] = (
        df[regression_columns]
        .mean(axis=1)
    )

    # Median is more robust against one model behaving badly.
    df["ensemble_return_median"] = (
        df[regression_columns]
        .median(axis=1)
    )

    # --------------------------------------------------------
    # Direction ensemble
    # --------------------------------------------------------

    direction_columns = [
        f"pred_direction_{model}"
        for model in DIRECTION_MODELS
    ]

    df["up_votes"] = (
        df[direction_columns]
        .sum(axis=1)
    )

    df["model_agreement"] = (
        df["up_votes"] /
        len(DIRECTION_MODELS)
    )

    # Probability-like confidence based on agreement.
    df["confidence"] = df["model_agreement"]

    # --------------------------------------------------------
    # Signal
    # --------------------------------------------------------

    df["signal"] = "HOLD"

    buy_condition = (
        (df["up_votes"] >= MIN_MODEL_AGREEMENT)
        &
        (df["confidence"] >= MIN_CONFIDENCE)
        &
        (df["ensemble_return"] >= MIN_EXPECTED_RETURN)
    )

    sell_condition = (
        (df["up_votes"] <= 1)
        &
        (df["ensemble_return"] < 0)
    )

    df.loc[buy_condition, "signal"] = "BUY"

    df.loc[sell_condition, "signal"] = "SELL"

    return df


# ============================================================
# RANK OPPORTUNITIES
# ============================================================

def rank_opportunities(
    df: pd.DataFrame,
) -> pd.DataFrame:

    df = df.copy()

    # Only consider BUY candidates.
    candidates = df[
        df["signal"] == "BUY"
    ].copy()

    if candidates.empty:
        df["selected"] = False
        df["position_weight"] = 0.0
        return df

    # --------------------------------------------------------
    # Opportunity score
    # --------------------------------------------------------

    candidates["opportunity_score"] = (
        candidates["ensemble_return"] * 0.60
        +
        candidates["confidence"] * 0.25
        +
        candidates["model_agreement"] * 0.15
    )

    # Rank within each trading day.
    candidates["rank"] = (
        candidates
        .groupby("date")["opportunity_score"]
        .rank(
            ascending=False,
            method="first",
        )
    )

    candidates["selected"] = (
        candidates["rank"] <= MAX_POSITIONS
    )

    # --------------------------------------------------------
    # Equal starting weight
    # --------------------------------------------------------

    selected = candidates[
        candidates["selected"]
    ].copy()

    if not selected.empty:

        selected["position_weight"] = (
            1.0 /
            selected.groupby("date")["ticker"]
            .transform("count")
        )

        selected["position_weight"] = (
            selected["position_weight"]
            .clip(
                upper=MAX_POSITION_WEIGHT
            )
        )

        # Re-normalize.
        totals = (
            selected
            .groupby("date")["position_weight"]
            .transform("sum")
        )

        selected["position_weight"] = (
            selected["position_weight"] /
            totals
        )

    # Merge results back.
    df["selected"] = False
    df["position_weight"] = 0.0
    df["opportunity_score"] = np.nan
    df["rank"] = np.nan

    if not selected.empty:

        key_columns = [
            "ticker",
            "date",
        ]

        selected_index = selected.set_index(
            key_columns
        )

        original_index = df.set_index(
            key_columns
        ).index

        common = original_index.intersection(
            selected_index.index
        )

        for idx in common:

            df.loc[
                (df["ticker"] == idx[0])
                &
                (df["date"] == idx[1]),
                "selected"
            ] = True

            df.loc[
                (df["ticker"] == idx[0])
                &
                (df["date"] == idx[1]),
                "position_weight"
            ] = selected_index.loc[
                idx,
                "position_weight"
            ]

            df.loc[
                (df["ticker"] == idx[0])
                &
                (df["date"] == idx[1]),
                "opportunity_score"
            ] = selected_index.loc[
                idx,
                "opportunity_score"
            ]

            df.loc[
                (df["ticker"] == idx[0])
                &
                (df["date"] == idx[1]),
                "rank"
            ] = selected_index.loc[
                idx,
                "rank"
            ]

    return df


# ============================================================
# STOCK LEVEL BACKTEST
# ============================================================

def calculate_stock_results(
    df: pd.DataFrame,
) -> pd.DataFrame:

    rows = []

    for ticker, group in df.groupby(
        "ticker"
    ):

        group = group.sort_values(
            "date"
        ).copy()

        capital = 1.0
        peak = 1.0
        max_drawdown = 0.0

        trades = 0
        wins = 0

        for _, row in group.iterrows():

            if not row["selected"]:
                continue

            predicted_return = float(
                row["ensemble_return"]
            )

            actual_return = float(
                row["target_return"]
            )

            # ------------------------------------------------
            # Stop loss
            # ------------------------------------------------

            strategy_return = actual_return

            if strategy_return <= -STOP_LOSS:
                strategy_return = -STOP_LOSS

            # ------------------------------------------------
            # Take profit
            # ------------------------------------------------

            if strategy_return >= TAKE_PROFIT:
                strategy_return = TAKE_PROFIT

            # Transaction cost.
            strategy_return -= TRANSACTION_COST

            capital *= (
                1.0 + strategy_return
            )

            trades += 1

            if strategy_return > 0:
                wins += 1

            peak = max(
                peak,
                capital,
            )

            drawdown = (
                capital / peak - 1
            )

            max_drawdown = min(
                max_drawdown,
                drawdown,
            )

        if trades == 0:
            continue

        rows.append(
            {
                "ticker": ticker,
                "final_growth": capital,
                "total_return": capital - 1,
                "max_drawdown": max_drawdown,
                "trades": trades,
                "win_rate": wins / trades,
            }
        )

    return pd.DataFrame(rows)


# ============================================================
# PORTFOLIO BACKTEST
# ============================================================

def portfolio_backtest(
    df: pd.DataFrame,
) -> tuple[pd.DataFrame, dict]:

    print("\n" + "=" * 70)
    print("PORTFOLIO BACKTEST")
    print("=" * 70)

    dates = sorted(
        df["date"].dropna().unique()
    )

    capital = INITIAL_CAPITAL

    daily_records = []

    peak_capital = capital

    total_trades = 0
    winning_days = 0
    trading_days = 0

    for date in dates:

        day = df[
            df["date"] == date
        ].copy()

        selected = day[
            day["selected"]
        ].copy()

        if selected.empty:

            daily_return = 0.0

        else:

            weights = (
                selected["position_weight"]
            )

            returns = (
                selected["target_return"]
            ).astype(float)

            daily_return = float(
                np.sum(
                    weights * returns
                )
            )

            # Transaction costs only for active positions.
            daily_return -= (
                TRANSACTION_COST *
                len(selected)
            )

            total_trades += len(selected)

            if daily_return > 0:
                winning_days += 1

            trading_days += 1

        capital *= (
            1.0 + daily_return
        )

        peak_capital = max(
            peak_capital,
            capital,
        )

        drawdown = (
            capital /
            peak_capital
            - 1.0
        )

        daily_records.append(
            {
                "date": date,
                "daily_return": daily_return,
                "capital": capital,
                "drawdown": drawdown,
                "positions": len(selected),
            }
        )

    daily = pd.DataFrame(
        daily_records
    )

    if daily.empty:
        raise RuntimeError(
            "Portfolio produced no daily results."
        )

    # ========================================================
    # RISK METRICS
    # ========================================================

    daily_returns = (
        daily["daily_return"]
        .astype(float)
    )

    total_return = (
        capital /
        INITIAL_CAPITAL
        - 1.0
    )

    annualized_return = (
        daily_returns.mean() * 252
    )

    annualized_volatility = (
        daily_returns.std() *
        np.sqrt(252)
    )

    if annualized_volatility > 0:

        sharpe = (
            annualized_return /
            annualized_volatility
        )

    else:
        sharpe = 0.0

    maximum_drawdown = (
        daily["drawdown"]
        .min()
    )

    win_rate = (
        (daily_returns > 0)
        .mean()
    )

    # CAGR.
    number_of_years = (
        len(daily) / 252
    )

    if number_of_years > 0:

        cagr = (
            capital /
            INITIAL_CAPITAL
        ) ** (
            1 / number_of_years
        ) - 1

    else:
        cagr = 0.0

    # Sortino.
    negative_returns = (
        daily_returns[
            daily_returns < 0
        ]
    )

    downside_deviation = (
        negative_returns.std() *
        np.sqrt(252)
        if len(negative_returns) > 1
        else 0
    )

    if downside_deviation > 0:

        sortino = (
            annualized_return /
            downside_deviation
        )

    else:
        sortino = 0.0

    # Profit factor.
    positive = (
        daily_returns[
            daily_returns > 0
        ].sum()
    )

    negative = abs(
        daily_returns[
            daily_returns < 0
        ].sum()
    )

    if negative > 0:
        profit_factor = (
            positive / negative
        )
    else:
        profit_factor = np.inf

    metrics = {
        "Initial_Capital": INITIAL_CAPITAL,
        "Final_Capital": capital,
        "Total_Return": total_return,
        "CAGR": cagr,
        "Annualized_Return": annualized_return,
        "Annualized_Volatility": annualized_volatility,
        "Sharpe_Ratio": sharpe,
        "Sortino_Ratio": sortino,
        "Maximum_Drawdown": maximum_drawdown,
        "Win_Rate": win_rate,
        "Profit_Factor": profit_factor,
        "Trading_Days": trading_days,
        "Total_Trades": total_trades,
    }

    return daily, metrics


# ============================================================
# BASELINE BUY AND HOLD
# ============================================================

def calculate_equal_weight_baseline(
    df: pd.DataFrame,
) -> dict:

    daily = (
        df.groupby("date")[
            "target_return"
        ]
        .mean()
        .fillna(0)
    )

    capital = INITIAL_CAPITAL

    equity = []

    for date, ret in daily.items():

        capital *= (
            1.0 + float(ret)
        )

        equity.append(
            capital
        )

    if not equity:
        return {
            "Baseline_Final_Capital":
                INITIAL_CAPITAL,
            "Baseline_Return": 0.0,
        }

    final_capital = equity[-1]

    return {
        "Baseline_Final_Capital":
            final_capital,

        "Baseline_Return":
            final_capital /
            INITIAL_CAPITAL
            - 1,
    }


# ============================================================
# SIGNAL SUMMARY
# ============================================================

def create_signal_summary(
    df: pd.DataFrame,
) -> pd.DataFrame:

    signal_df = df[
        df["selected"]
    ].copy()

    if signal_df.empty:
        return pd.DataFrame(
            columns=[
                "date",
                "ticker",
                "signal",
                "confidence",
                "ensemble_return",
                "opportunity_score",
                "position_weight",
            ]
        )

    columns = [
        "date",
        "ticker",
        "signal",
        "confidence",
        "ensemble_return",
        "opportunity_score",
        "position_weight",
        "up_votes",
    ]

    return (
        signal_df[columns]
        .sort_values(
            ["date", "opportunity_score"],
            ascending=[False, False],
        )
    )


# ============================================================
# MAIN
# ============================================================

def main():

    print("\n")
    print("=" * 70)
    print("PROFITABILITY STRATEGY ENGINE")
    print("=" * 70)

    # --------------------------------------------------------
    # 1. Load predictions
    # --------------------------------------------------------

    df = load_predictions()

    print(
        f"Date range : "
        f"{df['date'].min().date()} "
        f"to "
        f"{df['date'].max().date()}"
    )

    print(
        f"Stocks     : "
        f"{df['ticker'].nunique()}"
    )

    # --------------------------------------------------------
    # 2. Ensemble
    # --------------------------------------------------------

    print("\n" + "=" * 70)
    print("STEP 1 — MODEL ENSEMBLE")
    print("=" * 70)

    df = create_ensemble_signal(df)

    print(
        "Average predicted return:",
        f"{df['ensemble_return'].mean():.6f}"
    )

    print(
        "Average model agreement:",
        f"{df['model_agreement'].mean():.4f}"
    )

    # --------------------------------------------------------
    # 3. Rank opportunities
    # --------------------------------------------------------

    print("\n" + "=" * 70)
    print("STEP 2 — OPPORTUNITY SELECTION")
    print("=" * 70)

    df = rank_opportunities(df)

    selected_count = int(
        df["selected"].sum()
    )

    print(
        "Selected opportunities:",
        selected_count
    )

    print(
        "Unique selected stocks:",
        df.loc[
            df["selected"],
            "ticker"
        ].nunique()
    )

    # --------------------------------------------------------
    # 4. Stock results
    # --------------------------------------------------------

    print("\n" + "=" * 70)
    print("STEP 3 — STOCK PERFORMANCE")
    print("=" * 70)

    stock_results = calculate_stock_results(
        df
    )

    if not stock_results.empty:

        stock_results = (
            stock_results
            .sort_values(
                "total_return",
                ascending=False,
            )
        )

        print(
            stock_results.head(20)
            .to_string(index=False)
        )

    else:

        print(
            "No stocks passed the strategy filters."
        )

    # --------------------------------------------------------
    # 5. Portfolio
    # --------------------------------------------------------

    daily, metrics = portfolio_backtest(
        df
    )

    # --------------------------------------------------------
    # 6. Baseline
    # --------------------------------------------------------

    baseline = (
        calculate_equal_weight_baseline(
            df
        )
    )

    metrics.update(
        baseline
    )

    # --------------------------------------------------------
    # 7. Save outputs
    # --------------------------------------------------------

    REPORT_DIR.mkdir(
        parents=True,
        exist_ok=True,
    )

    stock_results.to_csv(
        OUTPUT_STOCK,
        index=False,
    )

    daily.to_csv(
        OUTPUT_DAILY,
        index=False,
    )

    signal_summary = (
        create_signal_summary(df)
    )

    signal_summary.to_csv(
        OUTPUT_SIGNALS,
        index=False,
    )

    metrics_df = pd.DataFrame(
        [metrics]
    )

    metrics_df.to_csv(
        OUTPUT_METRICS,
        index=False,
    )

    # --------------------------------------------------------
    # 8. Final report
    # --------------------------------------------------------

    print("\n")
    print("=" * 70)
    print("PROFITABILITY STRATEGY RESULTS")
    print("=" * 70)

    for key, value in metrics.items():

        if isinstance(value, float):

            if (
                "Return" in key
                or "CAGR" in key
                or "Drawdown" in key
                or "Rate" in key
                or "Volatility" in key
            ):

                print(
                    f"{key:28s}: "
                    f"{value:.2%}"
                )

            else:

                print(
                    f"{key:28s}: "
                    f"{value:.4f}"
                )

        else:

            print(
                f"{key:28s}: "
                f"{value}"
            )

    print("\n" + "=" * 70)
    print("FILES GENERATED")
    print("=" * 70)

    print(
        "Stock results :",
        OUTPUT_STOCK,
    )

    print(
        "Daily results :",
        OUTPUT_DAILY,
    )

    print(
        "Metrics       :",
        OUTPUT_METRICS,
    )

    print(
        "Signals       :",
        OUTPUT_SIGNALS,
    )

    print("\n" + "=" * 70)
    print("PROFITABILITY STRATEGY COMPLETE")
    print("=" * 70)


if __name__ == "__main__":
    main()
